```python
"""Single source of truth for wall clock. Tests freeze it via monkeypatch."""

from datetime import date, datetime
from threading import Lock

_lock = Lock()
_frozen: date | None = None


def today() -> date:
    """Return the process's notion of today. Frozen by frozen_clock fixture."""
    with _lock:
        if _frozen is not None:
            return _frozen
    return date.today()


def now() -> datetime:
    """Wall clock as datetime. Frozen to today's 09:00 to match the sample data_updated_at."""
    d = today()
    return datetime(d.year, d.month, d.day, 9, 0)


def freeze(d: date) -> None:
    """Set today's date for the process. Used by frozen_clock fixture."""
    global _frozen
    with _lock:
        _frozen = d


def unfreeze() -> None:
    global _frozen
    with _lock:
        _frozen = None
```

```python
"""Golden Set YAML loader and intent diff helper."""

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import date
from pathlib import Path
from typing import Any, Literal

import yaml

from app.intent.schema import (
    ComparisonKind,
    FilterCondition,
    FilterOperator,
    IntentKind,
    QueryIntent,
    TimeGrain,
    TimeRange,
)


_DEFAULT_AS_OF = date(2026, 8, 12)


# --- Dataclasses -----------------------------------------------------------


@dataclass(frozen=True)
class CitationExpectation:
    kind: str
    text: str | None = None


@dataclass(frozen=True)
class PolicySpec:
    kind: Literal["row_policy", "column_policy"]
    field: str
    allowed_values: tuple[str, ...] = ()
    masked: bool = False


@dataclass(frozen=True)
class FollowupSpec:
    as_user: str
    select_option_index: int
    expect: Expectation


@dataclass(frozen=True)
class IntentExpectation:
    metric: str | None = None
    time: TimeRange | None = None
    dimension: tuple[str, ...] = ()
    filters: tuple[FilterCondition, ...] = ()
    comparison: ComparisonKind | None = None
    top_n: int | None = None


@dataclass(frozen=True)
class Expectation:
    status: Literal["ANSWERED", "CLARIFYING", "REFUSED", "FAILED"]
    intent: IntentExpectation | None = None
    rows: int | None = None
    first_row: dict[str, Any] | None = None
    citation_has: tuple[CitationExpectation, ...] = ()
    refused_leaks: tuple[str, ...] = ()
    intent_diff: Literal["xfail", "fail"] = "xfail"
    clarify_kind: str | None = None


@dataclass(frozen=True)
class GoldenCase:
    id: str
    question: str
    as_user: str
    expect: Expectation
    as_of: date = _DEFAULT_AS_OF
    mode_required: Literal["any", "real_only", "stub_only"] = "any"
    policies: tuple[PolicySpec, ...] = ()
    followup: FollowupSpec | None = None
    notes: str = ""

    @property
    def has_first_expect(self) -> bool:
        """For followup cases, the first-turn expectation lives in `expect`."""
        return self.followup is not None


# --- Loaders ----------------------------------------------------------------


def _parse_date(s: str | date) -> date:
    if isinstance(s, date):
        return s
    return date.fromisoformat(s)


def _parse_filter(d: dict[str, Any]) -> FilterCondition:
    op = FilterOperator(d["operator"])
    values = tuple(d.get("values", ()))
    spoken_values = tuple(d.get("spoken_values", ()))
    return FilterCondition(
        field=d["field"],
        operator=op,
        values=values,
        spoken_values=spoken_values,
    )


def _parse_time_range(d: dict[str, Any] | None) -> TimeRange | None:
    if d is None:
        return None
    return TimeRange(
        start=_parse_date(d["start"]),
        end=_parse_date(d["end"]),
        grain=TimeGrain(d.get("grain", "DAY")),
        expression=d.get("expression", ""),
    )


def _parse_intent_expectation(d: dict[str, Any] | None) -> IntentExpectation | None:
    if d is None:
        return None
    return IntentExpectation(
        metric=d.get("metric"),
        time=_parse_time_range(d.get("time")),
        dimension=tuple(d.get("dimension", ())),
        filters=tuple(_parse_filter(f) for f in d.get("filters", ())),
        comparison=ComparisonKind(d["comparison"]) if d.get("comparison") else None,
        top_n=d.get("top_n"),
    )


def _parse_expectation(d: dict[str, Any]) -> Expectation:
    intent = _parse_intent_expectation(d.get("intent"))
    citation = tuple(
        CitationExpectation(kind=c["kind"], text=c.get("text"))
        for c in d.get("citation_has", ())
    )
    return Expectation(
        status=d["status"],
        intent=intent,
        rows=d.get("rows"),
        first_row=d.get("first_row"),
        citation_has=citation,
        refused_leaks=tuple(d.get("refused_leaks", ())),
        intent_diff=d.get("intent_diff", "xfail"),
        clarify_kind=d.get("clarify_kind"),
    )


def _parse_policy(d: dict[str, Any]) -> PolicySpec:
    return PolicySpec(
        kind=d["kind"],
        field=d["field"],
        allowed_values=tuple(d.get("allowed_values", ())),
        masked=bool(d.get("masked", False)),
    )


def _parse_followup(d: dict[str, Any] | None) -> FollowupSpec | None:
    if d is None:
        return None
    return FollowupSpec(
        as_user=d["as_user"],
        select_option_index=int(d["select_option_index"]),
        expect=_parse_expectation(d["expect"]),
    )


def _parse_golden_case(d: dict[str, Any]) -> GoldenCase:
    as_of_s = d.get("as_of")
    as_of = _parse_date(as_of_s) if as_of_s else _DEFAULT_AS_OF
    return GoldenCase(
        id=d["id"],
        question=d["question"],
        as_user=d["as_user"],
        expect=_parse_expectation(d["expect"]),
        as_of=as_of,
        mode_required=d.get("mode_required", "any"),
        policies=tuple(_parse_policy(p) for p in d.get("policies", ())),
        followup=_parse_followup(d.get("followup")),
        notes=d.get("notes", ""),
    )


def load_cases(cases_dir: Path) -> list[GoldenCase]:
    """Recursively load all *.yaml under cases_dir. Validate schema strictly."""
    cases: list[GoldenCase] = []
    seen_ids: set[str] = set()
    for yaml_path in sorted(cases_dir.rglob("*.yaml")):
        with yaml_path.open(encoding="utf-8") as f:
            raw = yaml.safe_load(f)
        if raw is None:
            continue
        # A single file may contain either a dict or a list of dicts.
        items = raw if isinstance(raw, list) else [raw]
        for item in items:
            case = _parse_golden_case(item)
            if case.id in seen_ids:
                raise ValueError(
                    f"duplicate Golden Set case id {case.id!r} in {yaml_path}"
                )
            seen_ids.add(case.id)
            cases.append(case)
    return cases


# --- Diff -------------------------------------------------------------------


def diff_in(
    actual: Any,
    expected: IntentExpectation,
) -> dict[str, tuple[Any, Any]]:
    """Return non-empty dict iff any expected slot mismatches actual.

    Slots with `None` value in expected are ignored.
    """
    diffs: dict[str, tuple[Any, Any]] = {}
    if expected.metric is not None and getattr(actual, "metric", None) != expected.metric:
        diffs["metric"] = (expected.metric, getattr(actual, "metric", None))
    if expected.time is not None:
        actual_time = getattr(actual, "time", None)
        if (
            actual_time is None
            or actual_time.start != expected.time.start
            or actual_time.end != expected.time.end
            or actual_time.grain != expected.time.grain
        ):
            diffs["time"] = (
                (expected.time.start, expected.time.end, expected.time.grain),
                (
                    getattr(actual_time, "start", None),
                    getattr(actual_time, "end", None),
                    getattr(actual_time, "grain", None),
                ),
            )
    if expected.dimension:
        actual_dim = tuple(getattr(actual, "dimension", ()))
        if actual_dim != expected.dimension:
            diffs["dimension"] = (expected.dimension, actual_dim)
    if expected.filters:
        actual_filters = tuple(getattr(actual, "filters", ()))
        if actual_filters != expected.filters:
            diffs["filters"] = (expected.filters, actual_filters)
    if expected.comparison is not None:
        if getattr(actual, "comparison", None) != expected.comparison:
            diffs["comparison"] = (
                expected.comparison,
                getattr(actual, "comparison", None),
            )
    if expected.top_n is not None:
        if getattr(actual, "top_n", None) != expected.top_n:
            diffs["top_n"] = (expected.top_n, getattr(actual, "top_n", None))
    return diffs
```

```python
"""Golden Set runner: executes one case, asserts the outcome, returns a report."""

from __future__ import annotations

import time
from dataclasses import dataclass
from typing import Any, Callable, Literal


@dataclass(frozen=True)
class CaseReport:
    id: str
    status: Literal["PASS", "XFAIL", "FAIL", "ERROR", "SKIPPED"]
    mode: Literal["stub", "real"]
    duration_ms: int
    message: str = ""
    diff: dict[str, tuple[Any, Any]] | None = None


def assert_case(outcome: dict[str, Any], expect: Expectation) -> AssertionError | None:
    """Run all assertions on outcome vs expect. Return AssertionError on first fail, else None."""
    actual_status = outcome.get("status")
    if actual_status != expect.status:
        return AssertionError(
            f"status mismatch: expected {expect.status!r}, got {actual_status!r}"
        )

    if expect.rows is not None:
        rows = outcome.get("rows") or []
        if len(rows) != expect.rows:
            return AssertionError(
                f"rows mismatch: expected {expect.rows} row(s), got {len(rows)}"
            )

    if expect.first_row is not None:
        rows = outcome.get("rows") or []
        if not rows:
            return AssertionError("first_row expected but rows is empty")
        first = rows[0]
        for key, expected_val in expect.first_row.items():
            if first.get(key) != expected_val:
                return AssertionError(
                    f"first_row[{key!r}] mismatch: expected {expected_val!r}, got {first.get(key)!r}"
                )

    if expect.citation_has:
        actual_kinds = {c.get("kind") for c in (outcome.get("citation") or [])}
        for c in expect.citation_has:
            if c.kind not in actual_kinds:
                return AssertionError(
                    f"citation_has missing kind {c.kind!r}: got {sorted(actual_kinds)}"
                )

    if expect.refused_leaks:
        text = outcome.get("response_text", "") or ""
        for token in expect.refused_leaks:
            if token in text:
                return AssertionError(
                    f"refusal leaks metadata: {token!r} found in {text!r}"
                )

    return None


def run_case(
    case: GoldenCase,
    *,
    mode: Literal["stub", "real"],
    orchestrator: Callable[..., dict[str, Any]],
    user_id_resolver: Callable[[str], int],
    ephemeral_policy: Callable[[str, tuple], None],
) -> CaseReport:
    """Execute one Golden Set case, return a CaseReport."""
    started = time.perf_counter()

    if case.mode_required == "real_only" and mode == "stub":
        return CaseReport(id=case.id, status="SKIPPED", mode=mode, duration_ms=0,
                          message="mode_required=real_only")
    if case.mode_required == "stub_only" and mode == "real":
        return CaseReport(id=case.id, status="SKIPPED", mode=mode, duration_ms=0,
                          message="mode_required=stub_only")

    try:
        if case.policies:
            ephemeral_policy(case.as_user, case.policies)
        user_id = user_id_resolver(case.as_user)
        client = _build_client(case, mode)
        outcome = orchestrator(
            question=case.question,
            user_id=user_id,
            client=client,
            llm_mode=mode,
        )
    except Exception as exc:
        return CaseReport(
            id=case.id, status="ERROR", mode=mode,
            duration_ms=int((time.perf_counter() - started) * 1000),
            message=f"{type(exc).__name__}: {exc}",
        )

    # Intent diff in real mode.
    if mode == "real" and case.expect.intent is not None:
        intent_diff = diff_in(outcome.get("intent"), case.expect.intent)
        if intent_diff:
            if case.expect.intent_diff == "xfail":
                return CaseReport(
                    id=case.id, status="XFAIL", mode=mode,
                    duration_ms=int((time.perf_counter() - started) * 1000),
                    diff=intent_diff,
                )
            return CaseReport(
                id=case.id, status="FAIL", mode=mode,
                duration_ms=int((time.perf_counter() - started) * 1000),
                diff=intent_diff,
                message="real-mode intent mismatch and intent_diff=fail",
            )

    err = assert_case(outcome, case.expect)
    elapsed = int((time.perf_counter() - started) * 1000)
    if err is not None:
        return CaseReport(id=case.id, status="FAIL", mode=mode,
                          duration_ms=elapsed, message=str(err))
    return CaseReport(id=case.id, status="PASS", mode=mode, duration_ms=elapsed)


def _build_client(case: GoldenCase, mode: str) -> Any:
    """Build the LLM client for this case.

    In stub mode, return StubClient fed with case.expect.intent.
    In real mode, return the singleton OpenAiCompatClient from settings.

    NOTE: actual classes come from plan 04. This helper assumes they exist
    and imports them lazily to avoid breaking the loader/runner test isolation.
    """
    if mode == "stub":
        from app.pipeline.stub_client import StubClient
        return StubClient(intent_payload=case.expect.intent)
    from app.pipeline.llm_client import OpenAiCompatClient
    from app.core.config import get_settings
    return OpenAiCompatClient(get_settings())
```

```python
from __future__ import annotations

from collections.abc import Callable, Iterator
from datetime import date
from typing import Any

import pytest

from app.core import clock
from tests.golden.loader import PolicySpec

FROZEN_TODAY = date(2026, 8, 12)


@pytest.fixture(autouse=True)
def frozen_clock() -> Iterator[None]:
    """Lock app.core.clock.today() to 2026-08-12 for the duration of each test.

    Teardown unfreezes so subsequent tests (or other conftest scopes) see the
    real date.
    """
    clock.freeze(FROZEN_TODAY)
    try:
        yield
    finally:
        clock.unfreeze()


@pytest.fixture
def ephemeral_policy(meta_session: Any) -> Iterator[Callable[[str, tuple[PolicySpec, ...]], None]]:
    """Install temporary row/column policies for a user, auto-clean on teardown.

    Returns an `install(user_name, policies)` callable. Cleanup runs in reverse
    insertion order.

    NOTE: The actual `insert_temporary_policy` / `delete_policy` calls come from
    plan 03 Task 1 (app.security.policy.repository). This conftest assumes the
    signatures and adapts if they differ at implementation time.
    """
    from app.security.policy.repository import (  # noqa: WPS433  (lazy import)
        delete_policy,
        insert_temporary_policy,
        resolve_user_id,
    )

    inserted_ids: list[int] = []

    def install(user_name: str, policies: tuple[PolicySpec, ...]) -> None:
        user_id = resolve_user_id(meta_session, user_name)
        for spec in policies:
            row_id = insert_temporary_policy(
                meta_session,
                user_id=user_id,
                kind=spec.kind,
                field=spec.field,
                allowed_values=list(spec.allowed_values),
                masked=spec.masked,
            )
            inserted_ids.append(row_id)

    yield install

    for row_id in reversed(inserted_ids):
        delete_policy(meta_session, row_id)
```

```python
def run_clarify_followup(
    case: GoldenCase,
    *,
    mode: Literal["stub", "real"],
    orchestrator: Callable[..., dict[str, Any]],
    user_id_resolver: Callable[[str], int],
    ephemeral_policy: Callable[[str, tuple], None],
) -> tuple[CaseReport, CaseReport]:
    """Execute a followup case: first turn asks the question, expects a
    clarification; second turn submits the chosen option, expects an answer.

    Returns (first_report, second_report). Both turns are executed even if
    the first one fails.
    """
    first_report = run_case(
        case,
        mode=mode,
        orchestrator=orchestrator,
        user_id_resolver=user_id_resolver,
        ephemeral_policy=ephemeral_policy,
    )

    # Build a synthesized follow-up GoldenCase with the same shape but no
    # recursive followup. The runner picks up the clarified answer via the
    # orchestrator's slot_state mechanism (plan 04 Task 7).
    from tests.golden.loader import GoldenCase as _GoldenCase  # noqa: WPS433

    followup_case = _GoldenCase(
        id=case.id + "-followup",
        question=case.question,  # orchestrator receives slot_state from first turn
        as_user=case.followup.as_user,
        expect=case.followup.expect,
        as_of=case.as_of,
        mode_required=case.mode_required,
        policies=case.policies,
        followup=None,
        notes=f"followup of {case.id}, select_option_index={case.followup.select_option_index}",
    )

    second_report = run_case(
        followup_case,
        mode=mode,
        orchestrator=orchestrator,
        user_id_resolver=user_id_resolver,
        ephemeral_policy=ephemeral_policy,
    )

    return first_report, second_report
```

```python
"""pytest terminal summary hook for Golden Set reports."""

from __future__ import annotations

from typing import Any


def pytest_terminal_summary(terminalreporter: Any) -> None:
    """Print the Golden Set summary block at the end of the test session.

    The hook reads from terminalreporter.stats which is keyed by report outcome.
    """
    # Collect by (mode, status) → count
    counts: dict[tuple[str, str], int] = {}
    for status_key, reports in terminalreporter.stats.items():
        if not reports:
            continue
        for report in reports:
            when = getattr(report, "when", None)
            if when != "call":
                continue
            case_id = getattr(report, "nodeid", "")
            if "tests/golden/" not in case_id:
                continue
            # Mode is encoded in the case parametrize id (e.g. "G-001-stub")
            # or in the case.id suffix. Default to "stub" if not present.
            mode = "stub"
            for m in ("stub", "real"):
                if m in case_id:
                    mode = m
                    break
            outcome_key = status_key  # 'passed', 'failed', 'skipped'
            outcome = {
                "passed": "Pass",
                "failed": "Fail",
                "error": "Error",
                "skipped": "Skipped",
                "xfailed": "Xfail",
            }.get(outcome_key, outcome_key)
            counts[(mode, outcome)] = counts.get((mode, outcome), 0) + 1

    if not counts:
        return

    terminalreporter.write_sep("=", "Golden Set Summary")
    for mode in ("stub", "real"):
        line = (
            f"Mode: {mode:<5} | "
            f"Pass: {counts.get((mode, 'Pass'), 0):<3} | "
            f"Xfail: {counts.get((mode, 'Xfail'), 0):<3} | "
            f"Fail: {counts.get((mode, 'Fail'), 0):<3} | "
            f"Error: {counts.get((mode, 'Error'), 0):<3} | "
            f"Skipped: {counts.get((mode, 'Skipped'), 0):<3}"
        )
        terminalreporter.write_line(line)
    terminalreporter.write_sep("=", "Golden Set Summary")
```
